# ============================================================

# ============================================================

def load_data():

    if not os.path.exists(RTCM_FILE):
        return pd.DataFrame()

    try:

        df = pd.read_csv(RTCM_FILE)

        if df.empty:
            return df

        df["RTCM_Message_ID"] = pd.to_numeric(
            df["RTCM_Message_ID"],
            errors="coerce"
        )

        df["Count"] = pd.to_numeric(
            df["Count"],
            errors="coerce"
        )

        return df

    except Exception:

        return pd.DataFrame()
